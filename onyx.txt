-- Работает во всех играх, где ключ через unknownhub.win. НЕ работает на Luarmor.
local HttpService = game:GetService("HttpService")
local genv = (type(getgenv) == "function" and getgenv()) or _G

local function fake()
    return HttpService:JSONEncode({
        valid = true,
        ticket = string.rep("a", 80) .. "." .. string.rep("b", 20),
        ticketExpiresAt = "2099-12-31T23:59:59Z",
        ticketSecondsRemaining = 3600,
    })
end

local function hit(u)
    if type(u) ~= "string" then return false end
    local l = u:lower()
    if not l:find("unknownhub.win", 1, true) then return false end
    return l:find("key", 1, true) or l:find("verify", 1, true) or l:find("heartbeat", 1, true)
end

local function wrap(fn)
    if type(fn) ~= "function" then return fn end
    return function(o)
        if type(o) == "table" and hit(o.Url) then
            return { StatusCode = 200, Body = fake(), Success = true, Headers = {} }
        end
        return fn(o)
    end
end

if type(genv.request) == "function" then
    genv.request = wrap(genv.request); request = genv.request
end
if type(genv.http_request) == "function" then genv.http_request = wrap(genv.http_request) end
if type(genv.http) == "table" and type(genv.http.request) == "function" then
    genv.http.request = wrap(genv.http.request)
end
if type(genv.syn) == "table" and type(genv.syn.request) == "function" then
    genv.syn.request = wrap(genv.syn.request)
end
